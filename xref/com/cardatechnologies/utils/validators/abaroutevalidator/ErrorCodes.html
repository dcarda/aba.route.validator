23  package com.cardatechnologies.utils.validators.abaroutevalidator;
24  
25  /**
26   * <b>Description:</b><br>
27   *     This enumeration defines the error codes and associated message
28   *     identifiers used by the ABA Routing Transit Number validator.
29   *
30   * @author     Daniel Carda
31   * <br>
32   * <br><b>Maintenance History:</b>
33   * <br>
34  <pre>
35      yyyy mm dd  Who                       Description
36      ----------  ------------------------  ----------------------------------------------------
37      2015/12/22  Daniel Carda              Initial Module Creation...
38      2020/12/20  Daniel Carda              Updated formatting.
39      2021/08/01  Daniel Carda              Improved Class Javadocs
40      2026/08/01  Daniel Carda              Improved Mnemonic Text.  New Class Header.
41  </pre>
42   * <hr>
43   */
44  public enum ErrorCodes {
45  
46      /** Error code indicating ABA number is null. */
47      ABA_1000(-1000,
48              "com.cardatechnologies.aba.number.null"),
49  
50      /** Error code indicating ABA number is blank. */
51      ABA_1001(-1001,
52              "com.cardatechnologies.aba.number.blank"),
53  
54      /** Error code indicating ABA number is too short. */
55      ABA_1002(-1002,
56              "com.cardatechnologies.aba.number.length.too.short"),
57  
58      /** Error code indicating ABA number is too long. */
59      ABA_1003(-1003,
60              "com.cardatechnologies.aba.number.length.too.long"),
61  
62      /** Error code indicating ABA number contains non-ASCII digits. */
63      ABA_1004(-1004,
64              "com.cardatechnologies.aba.number.contains.non.ascii.digits"),
65  
66      /** Error code indicating an invalid Federal Reserve prefix. */
67      ABA_1005(-1005,
68              "com.cardatechnologies.aba.number.invalid.federal.reserve.prefix");
69  
70      private final int    errorCode;
71      private final String errorMnemonic;
72  
73      /**
74       * Constructs ...
75       *
76       * @param errorCode
77       *        Integer value for the error code
78       * @param errorMnemonic
79       *        Text mnemonic describing the error code.
80       */
81      ErrorCodes( int    errorCode,
82                  String errorMnemonic ) {
83          this.errorCode     = errorCode;
84          this.errorMnemonic = errorMnemonic;
85      }
86  
87      /**
88       * Method: getErrorCode
89       * <p>
90       * <b>Description:</b><br>
91       * This will return the integer for the error code.
92       *
93       * @return int
94       */
95      public int getErrorCode() {
96          return errorCode;
97      }
98  
99      /**
100      * Method: getErrorMnemonic
101      * <p>
102      * <b>Description:</b><br>
103      * This will return the text of the error code.
104      *
105      * @return String
106      */
107     public String getErrorMnemonic() {
108         return errorMnemonic;
109     }
110 
111     /**
112      * Method: fromErrorCode
113      *
114      * <p>
115      * <b>Description:</b><br>
116      * Returns the ErrorCodes enumeration associated with the
117      * specified numeric error code.
118      *
119      * @param errorCode
120      *        The numeric error code to locate.
121      *
122      * @return The corresponding ErrorCodes enumeration.
123      *
124      * @throws IllegalArgumentException
125      *         If the specified error code does not exist.
126      */
127     public static ErrorCodes fromErrorCode(final int errorCode) {
128 
129         for (final ErrorCodes error : values()) {
130             if (error.getErrorCode() == errorCode) {
131                 return error;
132             }
133         }
134 
135         throw new IllegalArgumentException(
136                 "Unknown ABA routing validation error code: " + errorCode);
137     }
138 }
